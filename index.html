app.css
/* THE ARC IS SOLVED FOR, NOT PLACED.
the chord subtended by --pitch is
2r·sin(pitch/2); set that equal to
--gap diameters and solve for r */
.splay {
  --pitch: 46deg;
  --gap: 1.5;
  --r: calc(var(--d) * var(--gap)
       / (2 * sin(var(--pitch) / 2)));
}

.ch {
  --a: calc((var(--i) - (var(--n) - 1) / 2)
       * var(--pitch));
  transform:
    translate(
      calc(sin(var(--a)) * var(--r) * var(--t)),
      calc(cos(var(--a)) * var(--r) * var(--t) * -1)
    ) scale(var(--s));
}
